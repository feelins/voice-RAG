#### 1. ASR 演示脚本

"""
演示 1：语音识别（ASR）独立演示
展示如何：注册账号 → 获取 Key → 调用 API → 看到结果
"""

# ===== 步骤 1：配置 API Key =====
# 打开：https://dashscope.console.aliyun.com/
# 注册 → 开通服务 → 复制 API Key
API_KEY = "sk-xxxxxxxxxxxxxxxx"  # 现场填入

# ===== 步骤 2：安装依赖 =====
# pip install dashscope

# ===== 步骤 3：最简调用 =====
import dashscope
from dashscope.audio.asr import Recognition

dashscope.api_key = API_KEY

def test_asr(audio_file):
    """测试语音识别"""
    print(f"正在识别: {audio_file}")
    
    recognizer = Recognition(
        model='paraformer-realtime-v2',
        format='wav',
        sample_rate=16000
    )
    
    result = recognizer.call(audio_file)
    
    if result.status_code == 200:
        text = result.output.get('text', '')
        print(f"✅ 识别结果: {text}")
        return text
    else:
        print(f"❌ 识别失败: {result.message}")
        return None

# ===== 步骤 4：运行测试 =====
if __name__ == '__main__':
    # 使用预先录制的测试音频
    test_audio = 'test_audio/question1.wav'
    test_asr(test_audio)
    
    # 输出示例：
    # 正在识别: test_audio/question1.wav
    # ✅ 识别结果: 学而时习之是什么意思
```

#### 2. TTS 演示脚本

"""
演示 2：语音合成（TTS）独立演示
"""

import dashscope
from dashscope.audio.tts_v2 import SpeechSynthesizer

API_KEY = "sk-xxxxxxxxxxxxxxxx"  # 使用同一个 Key
dashscope.api_key = API_KEY

def test_tts(text, output_file='output.mp3'):
    """测试语音合成"""
    print(f"正在合成: {text}")
    
    synthesizer = SpeechSynthesizer(
        model='cosyvoice-v1',
        voice='longxiaochun'  # 女声
    )
    
    audio_data = synthesizer.call(text)
    
    if isinstance(audio_data, bytes):
        with open(output_file, 'wb') as f:
            f.write(audio_data)
        print(f"✅ 合成成功: {output_file}")
        
        # 自动播放
        import os
        os.system(f'start {output_file}')  # Windows
        return True
    else:
        print(f"❌ 合成失败")
        return False

if __name__ == '__main__':
    # 测试不同音色
    voices = {
        'longxiaochun': '普通话-女声',
        'longyuan': '普通话-男声',
        'longjiaxin_v3': '粤语-女声'
    }
    
    text = "学而时习之，不亦说乎？"
    
    for voice_id, voice_name in voices.items():
        print(f"\n测试音色: {voice_name}")
        synthesizer = SpeechSynthesizer(model='cosyvoice-v1', voice=voice_id)
        audio = synthesizer.call(text)
        
        if isinstance(audio, bytes):
            filename = f'output_{voice_id}.mp3'
            with open(filename, 'wb') as f:
                f.write(audio)
            print(f"✅ 生成: {filename}")
```

**演示流程**（5分钟）：
1. 使用同一个 Key
2. 运行脚本，现场播放合成的语音
3. 对比不同音色效果

#### 3. 知识库演示
**文件**：`demos/03_kb_demo.py`
```python
"""
演示 3：两种知识库方案对比
"""

# ===== 方案 1：硬编码（最简单）=====
QA_DATABASE = {
    "学而时习之是什么意思": "学习了知识，并且时常温习实践，不也是很愉快的事吗？",
    "孔子的弟子有谁": "孔子有众多弟子，其中最著名的是「孔门十哲」...",
}

def find_answer_v1(question):
    """方案1：字典查询"""
    return QA_DATABASE.get(question, "抱歉，我不知道")

# ===== 方案 2：使用 LLM（智能）=====
def find_answer_v2(question):
    """方案2：调用通义千问"""
    from dashscope import Generation
    
    prompt = f"""你是古籍专家，回答问题：{question}
    
知识库：
{QA_DATABASE}

请基于知识库回答。"""
    
    response = Generation.call(
        model='qwen-turbo',
        prompt=prompt
    )
    
    if response.status_code == 200:
        return response.output.text
    else:
        return "查询失败"

# ===== 对比测试 =====
if __name__ == '__main__':
    questions = [
        "学而时习之是什么意思",      # 精确匹配
        "学而时习之讲的是什么",      # 模糊表达
        "孔子强调学习要怎么做",      # 需要理解
    ]
    
    for q in questions:
        print(f"\n问题: {q}")
        print(f"方案1（硬编码）: {find_answer_v1(q)}")
        print(f"方案2（LLM）: {find_answer_v2(q)}")
```

**演示流程**（10分钟）：
1. 先展示硬编码方案：简单但不灵活
2. 再展示 LLM 方案：智能但需要 prompt 工程
3. 引出 FastGPT：可视化配置，兼顾两者优点

### 第二层：模块整合演示

#### 整合脚本
**文件**：`demos/04_integration_demo.py`
```python
"""
演示 4：三模块整合（命令行版）
展示：ASR → 知识库 → TTS 的完整流程
"""

import dashscope
from dashscope.audio.asr import Recognition
from dashscope.audio.tts_v2 import SpeechSynthesizer

API_KEY = "sk-xxxxxxxxxxxxxxxx"
dashscope.api_key = API_KEY

# 知识库（简化）
QA_DB = {
    "学而时习之是什么意思": "学习了知识，并且时常温习实践，不也是很愉快的事吗？"
}

def asr(audio_file):
    """ASR 模块"""
    recognizer = Recognition(model='paraformer-realtime-v2', format='wav', sample_rate=16000)
    result = recognizer.call(audio_file)
    return result.output.get('text', '') if result.status_code == 200 else None

def kb(question):
    """知识库模块"""
    return QA_DB.get(question, "抱歉，我不知道")

def tts(text, output='answer.mp3'):
    """TTS 模块"""
    synthesizer = SpeechSynthesizer(model='cosyvoice-v1', voice='longxiaochun')
    audio_data = synthesizer.call(text)
    if isinstance(audio_data, bytes):
        with open(output, 'wb') as f:
            f.write(audio_data)
        return output
    return None

def main(audio_file):
    """完整流程"""
    print("=" * 50)
    print("古籍问答系统 - 命令行版")
    print("=" * 50)
    
    # 步骤 1: ASR
    print("\n[1/3] 正在识别语音...")
    question = asr(audio_file)
    if not question:
        print("❌ 识别失败")
        return
    print(f"✅ 识别结果: {question}")
    
    # 步骤 2: 知识库
    print("\n[2/3] 正在查询知识库...")
    answer = kb(question)
    print(f"✅ 答案: {answer}")
    
    # 步骤 3: TTS
    print("\n[3/3] 正在合成语音...")
    audio_file = tts(answer)
    if audio_file:
        print(f"✅ 语音已生成: {audio_file}")
        import os
        os.system(f'start {audio_file}')
    else:
        print("❌ 合成失败")
    
    print("\n" + "=" * 50)
    print("流程完成！")

if __name__ == '__main__':
    main('test_audio/question1.wav')
```

**演示流程**（10分钟）：
1. 运行脚本，观察三个步骤依次执行
2. 解释：这就是核心逻辑，不到 100 行
3. 引出：加上界面就是完整系统

## 总结

### 分层演示的优势
✅ **循序渐进**：从单模块到整合，易于理解  
✅ **实时操作**：Key 现场输入，演示真实流程  
✅ **可复现**：学员能跟着操作，不是"玄学"  
✅ **灵活调整**：时间不够可跳过进阶内容  

### 核心原则
1. **讲师演示**：Key 实时输入，不预置
2. **学员实操**：提供预配置环境，降低门槛
3. **逐步深入**：先跑通，再理解，最后优化
